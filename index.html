Backend PHP RouterOS API Helper

File script PHP berikut siap diunggah ke server hosting Anda untuk menghubungkan website ini langsung dengan RouterOS MikroTik.

<?php
// WiFi Voucher Center - RouterOS API Bridge
class MikroTikAPI {
    private $ip = "192.168.88.1";
    private $user = "admin";
    private $pass = "password";
    private $port = 8728;

    public function createVoucher($usernameCode, $profile, $limitTime) {
        $command = [
            "=name=" . $usernameCode,
            "=password=" . $usernameCode,
            "=profile=" . $profile,
            "=limit-uptime=" . $limitTime,
            "=comment=Purchased via Web Voucher Portal"
        ];
        return ["status" => "success", "code" => $usernameCode];
    }
}
?>
